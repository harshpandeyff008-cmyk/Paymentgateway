                POST
                /api/v1/orders
              

              Base URL: https://paypendicular.onrender.com
            


            

              Creates a new dynamic UPI payment request with unique paise collision protection, high-resolution Base64 QR code image, deep UPI intent links, and custom checkout branding.
            


            
            

              

                
                  

                    Parameter
                    Type
                    Required
                    Description
                  

                
                
                  

                    amount
                    Float
                    Required
                    Transaction amount in INR (e.g. 250.00). Auto-offset by unique paise (e.g. 250.04) for 1-second auto-match.
                  

                  

                    customerName
                    String
                    Optional
                    Customer's full name (e.g. "Rahul Verma").
                  

                  

                    customerPhone
                    String
                    Optional
                    Customer's 10-digit mobile number for WhatsApp delivery & tracking.
                  

                  

                    webhookUrl
                    URL
                    Optional
                    Your server endpoint that will receive a POST notification instantly upon bank settlement.
                  

                  

                    expiryMinutes
                    Integer
                    Optional
                    Order validity duration (Default: 20, Max: 1440 mins / 24 hours).
                  

                  

                    customNote
                    String
                    Optional
                    Custom note displayed on customer checkout stand (e.g. "Order #4029 - Instant License Key").
                  

                  

                    theme
                    String
                    Optional
                    Checkout stand theme: tiranga (default), cyan, purple, or dark.
                  

                  

                    redirectUrl
                    URL
                    Optional
                    Auto-redirect URL for customer after payment confirmation.

                Python (Requests)
                
              

              
import requests

url = "https://paypendicular.onrender.com/api/v1/orders"
headers = {
    "Content-Type": "application/json",
    "x-api-key": "pg_live_..."
}
payload = {
    "amount": 250.00,
    "customerName": "Rahul Verma",
    "customerPhone": "9876543210",
    "webhookUrl": "https://yourwebsite.com/api/payment-webhook"
}

response = requests.post(url, json=payload, headers=headers)
order = response.json().get("order")
print("Order Code:", order.get("orderCode"))
print("Payable Amount:", order.get("amount"))
print("Checkout URL:", order.get("checkoutUrl"))

<?php
$ch = curl_init('https://paypendicular.onrender.com/api/v1/orders');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Content-Type: application/json',
    'x-api-key: pg_live_...'
]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
    'amount' => 250.00,
    'customerName' => 'Rahul Verma',
    'customerPhone' => '9876543210',
    'webhookUrl' => 'https://yourwebsite.com/api/payment-webhook'
]));

$response = curl_exec($ch);
curl_close($ch);
$data = json_decode($response, true);
echo "Order: " . $data['order']['orderCode'] . " | Amount: " . $data['order']['amount'];
?>

});
